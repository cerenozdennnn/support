function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = sheet.getDataRange().getValues(); var result = []; for (var i = 1; i < data.length; i++) { result.push({ id: data[i][0], company: data[i][1], category: data[i][2], title: data[i][3], description: data[i][4], status: data[i][5], rca: data[i][6], timestamp: data[i][7] }); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var params = JSON.parse(e.postData.contents); if (params.action === "create") { sheet.appendRow([ params.id, params.company, params.category, params.title, params.description, params.status || "Açık", params.rca || "", new Date().toISOString() ]); } else if (params.action === "update") { var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (data[i][0] === params.id) { if (params.status !== undefined) sheet.getRange(i + 1, 6).setValue(params.status); if (params.rca !== undefined) sheet.getRange(i + 1, 7).setValue(params.rca); break; } } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); }
